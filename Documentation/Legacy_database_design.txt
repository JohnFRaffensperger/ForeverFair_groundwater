Table: auctions
  solve_time            DATETIME
  solve_status          TEXT
  auction_id            INTEGER  PK
  auction_type          varchar

  firstTradedDate       DATETIME

Table: control_point_event
  CPE_id                INTEGER  PK
  RMI_id                INTEGER  (FK -> response_matrix_info)
  CP_id                 INTEGER  (FK -> control_points)
  auction_id            INTEGER  (FK -> auctions)
  CPE_effect_date       DATETIME
  CPE_equality_constraint_right_hand_side INTEGER

  CPE_constraint_lower_bound INTEGER
  CPE_head_constraint_upper_bound INTEGER

  CPE_range_upper       INTEGER

Table: control_points
  CP_id                 INTEGER  PK
  CP_layer1             INTEGER
  CP_row1               INTEGER
  CP_col1               INTEGER
  CP_layer2             INTEGER
  CP_row2               INTEGER
  CP_col2               INTEGER
  CP_latitude1          INTEGER
  CP_longitude1         INTEGER
  CP_latitude2          INTEGER
  CP_longitude2         INTEGER
  CP_type               INTEGER
  CP_name               TEXT
  CP_description        TEXT
  NOTE: Two grid locations (layer/row/col 1 and 2) and two lat/lon pairs stored,
        presumably to support a two-point averaging or difference constraint.

Table: messages
  id                    INTEGER  PK
  message_type          TEXT
  message               TEXT

Table: metadata
  meta_key              TEXT  PK
  meta_value            TEXT  NN
  NOTE: Added post-2012 (not in original Design 2). Key-value store for
        catchment_name, rights policy, current trader, etc.

Table: response_matrix
  coefficient_id        INTEGER  PK
  RMI_id                INTEGER  (FK -> response_matrix_info)
  wellid                INTEGER  (FK -> wells)
  control_point_id      INTEGER  (FK -> control_points)
  RM_take_of_water_date DATETIME
  RM_effect_on_control_point_date INTEGER
  RM_impact_coefficient INTEGER

Table: response_matrix_info
  RMI_hours_period_length NUMERIC
  RMI_id                INTEGER  PK
  RMI_creation_date     DATETIME
  RMI_cubic_meters_per_impulse_unit INTEGER
  RMI_notes             TEXT

Table: traderbids
  well_id               NUMERIC
  bidid                 INTEGER  PK
  traderID              INTEGER  (FK -> traders)
  auctionID             INTEGER  (FK -> auctions)
  bidDate               DATETIME
  effectDate            DATETIME
  expiryDate            DATETIME
  isBidAutomatic        boolean
  qty1                  REAL
  price1                REAL
  qty2                  REAL
  price2                REAL
  qty3                  REAL
  price3                REAL
  qty4                  REAL
  price4                REAL
  qty5                  REAL
  price5                REAL
  NOTE: Up to 5 price-quantity bid steps stored as columns on a single row.

Table: traderconsents
  consentid             TEXT  PK
  traderID              TEXT  (FK -> traders)
  wellID                INTEGER  (FK -> wells)
  consentQuantity       REAL
  startDate             DATETIME
  expiryDate            DATETIME

Table: traderquota
  quotaid               INTEGER  PK
  traderid              INTEGER  (FK -> traders)
  auctionid             INTEGER  (FK -> auctions)
  wellID                INTEGER  (FK -> wells)
  quotaAuctionStart     REAL
  quotaAdjusted         REAL
  quotaAuctionEnd       REAL
  price                 REAL
  lprowName             TEXT
  takeDate              DATETIME
  NOTE: One row per well per auction per take-date. The three quota columns
        represent the three stages of quota lifecycle:
          quotaAuctionStart  - entitlement entering the auction (from consent)
          quotaAdjusted      - after proportional/alpha scaling when constraints bind
          quotaAuctionEnd    - final usable quota after clearing

Table: traders
  traderid              INTEGER  PK
  trader_loginid        TEXT
  trader_password       TEXT
  trader_fname          TEXT
  trader_lname          TEXT
  trader_address        TEXT
  trader_city           TEXT
  trader_phone          TEXT
  trader_email          TEXT

Table: wells
  well_id               INTEGER  PK
  well_layer            INTEGER
  well_row              INTEGER
  well_column           INTEGER
  latitude              REAL
  longitude             REAL
  NOTE: latitude and longitude added post-2012 (not in original Design 2).

================================================================================
NOTES ON QUOTA TERMINOLOGY
================================================================================
Table traderquota explicitly names three quota stages:
  quotaAuctionStart  - entitlement at the start of the auction (from consent)
  quotaAdjusted      - entitlement after proportional/alpha scaling when
                       hydrologic constraints bind
  quotaAuctionEnd    - final usable quota after auction clearing
